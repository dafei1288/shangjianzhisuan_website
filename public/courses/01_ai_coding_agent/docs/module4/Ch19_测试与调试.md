## 核心概念

### 为什么 LLM Agent 必须用 Mock 测试

```
真实 LLM 调用的问题：
  - 输出不确定：相同输入可能产生不同输出 → 无法断言
  - 延迟高：每次调用 1-10 秒 → 测试套件运行太慢
  - 费用：每次测试都消耗 token → 持续集成成本高
  - 依赖网络：断网则测试全部失败 → CI 不稳定

Mock LLM 调用的优势：
  - 输出确定：每次返回预设的固定响应 → 可以精确断言
  - 零延迟：内存操作 → 测试秒级完成
  - 零费用：不调用真实 API → 无成本
  - 无网络依赖：纯本地运行 → CI 稳定
```

### 测试金字塔

```
        /  E2E 测试  \          ← 少量，真实 LLM，验证整体流程
       /  (真实 API)  \
      /────────────────\
     /  集成测试         \      ← 中等数量，Mock LLM，验证模块协作
    /  (Mock Provider)   \
   /──────────────────────\
  /   单元测试               \  ← 大量，Mock 依赖，验证单个函数
 /    (纯函数/工具函数)       \
/──────────────────────────────\
```

### 测试覆盖的关键路径

1. **工具函数**：read_file、write_file、edit_file、run_bash 的正常和异常路径
2. **Provider 层**：API 响应解析、流式处理、错误重试
3. **Agent Loop**：文本回复路径、工具调用路径、多轮工具调用、错误恢复
4. **上下文管理**：token 计数、预算检查、压缩触发
5. **安全层**：注入检测、命令检查、路径检查

### 19.1 MockFactory — Mock 响应工厂

```python
"""
MockFactory: 快速构造各种类型的 Mock LLM 响应。
支持文本回复、工具调用、流式输出等场景。
"""
import json
import time
from dataclasses import dataclass, field
from typing import Optional
from unittest.mock import MagicMock


@dataclass
class MockToolCall:
    """模拟的工具调用"""
    name: str
    arguments: dict
    call_id: str = ""

    def __post_init__(self):
        if not self.call_id:
            self.call_id = f"call_{self.name}_{int(time.time() * 1000)}"


class MockFactory:
    """
    Mock 响应工厂。
    用最少的代码构造各种类型的 LLM API 响应对象。
    """

    @staticmethod
    def text_response(content: str, finish_reason: str = "stop") -> MagicMock:
        """构造纯文本回复"""
        msg = MagicMock()
        msg.content = content
        msg.tool_calls = None
        msg.role = "assistant"

        choice = MagicMock()
        choice.message = msg
        choice.finish_reason = finish_reason
        choice.index = 0

        response = MagicMock()
        response.choices = [choice]
        response.id = "mock-response-id"
        response.usage = MagicMock(
            prompt_tokens=100,
            completion_tokens=50,
            total_tokens=150,
        )
        return response

    @staticmethod
    def tool_call_response(
        tool_calls: list[MockToolCall],
        content: Optional[str] = None,
    ) -> MagicMock:
        """构造工具调用响应（单工具或多工具）"""
        mock_tool_calls = []
        for tc in tool_calls:
            mock_tc = MagicMock()
            mock_tc.id = tc.call_id
            mock_tc.function.name = tc.name
            mock_tc.function.arguments = json.dumps(tc.arguments)
            mock_tool_calls.append(mock_tc)

        msg = MagicMock()
        msg.content = content
        msg.tool_calls = mock_tool_calls
        msg.role = "assistant"

        choice = MagicMock()
        choice.message = msg
        choice.finish_reason = "tool_calls"
        choice.index = 0

        response = MagicMock()
        response.choices = [choice]
        response.id = "mock-response-id"
        response.usage = MagicMock(
            prompt_tokens=200,
            completion_tokens=80,
            total_tokens=280,
        )
        return response

    @staticmethod
    def multi_turn_responses(
        turns: list[dict],
    ) -> list[MagicMock]:
        """
        构造多轮响应序列。
        每轮可以是 {"text": "..."} 或 {"tool_calls": [...]}。
        """
        responses = []
        for turn in turns:
            if "text" in turn:
                responses.append(
                    MockFactory.text_response(turn["text"])
                )
            elif "tool_calls" in turn:
                mock_calls = [
                    MockToolCall(**tc) if isinstance(tc, dict) else tc
                    for tc in turn["tool_calls"]
                ]
                responses.append(
                    MockFactory.tool_call_response(mock_calls)
                )
        return responses

    @staticmethod
    def stream_chunks(text: str, chunk_size: int = 5) -> list[MagicMock]:
        """构造流式输出的 chunk 序列"""
        chunks = []
        for i in range(0, len(text), chunk_size):
            chunk_text = text[i:i + chunk_size]

            delta = MagicMock()
            delta.content = chunk_text
            delta.role = None

            choice = MagicMock()
            choice.delta = delta
            choice.finish_reason = None

            chunk = MagicMock()
            chunk.choices = [choice]
            chunks.append(chunk)

        # 最后一个 chunk 表示结束
        final_delta = MagicMock()
        final_delta.content = ""
        final_delta.role = None

        final_choice = MagicMock()
        final_choice.delta = final_delta
        final_choice.finish_reason = "stop"

        final_chunk = MagicMock()
        final_chunk.choices = [final_choice]
        chunks.append(final_chunk)

        return chunks

    @staticmethod
    def error_response(status_code: int, message: str) -> Exception:
        """构造 API 错误响应"""
        error = MagicMock()
        error.status_code = status_code
        error.message = message

        api_error = MagicMock()
        api_error.response = MagicMock()
        api_error.response.status_code = status_code
        api_error.response.json.return_value = {
            "error": {"message": message, "type": "invalid_request_error"}
        }
        return api_error
```

"""

使用 tempfile 创建临时文件系统，测试后自动清理。
"""
import pytest
import tempfile
import os
from pathlib import Path


class TestReadFile:
    """read_file 工具的单元测试"""

    def test_read_normal_file(self):
        """测试读取普通文件"""
        with tempfile.NamedTemporaryFile(
            mode="w", suffix=".py", delete=False, encoding="utf-8"
        ) as f:
            f.write("def hello():\n    return 'world'\n")
            path = f.name

        try:
            result = read_file(path)
            assert "def hello():" in result
            assert "return 'world'" in result
        finally:
            os.unlink(path)

    def test_read_nonexistent_file(self):
        """测试读取不存在的文件"""
        result = read_file("/nonexistent/path/file.py")
        assert "Error" in result or "不存在" in result

    def test_read_with_line_range(self):
        """测试带行号范围的读取"""
        with tempfile.NamedTemporaryFile(
            mode="w", suffix=".txt", delete=False
        ) as f:
            for i in range(1, 101):
                f.write(f"Line {i}\n")
            path = f.name

        try:
            result = read_file(path, start_line=10, end_line=20)
            assert "Line 10" in result
            assert "Line 20" in result
            assert "Line 1" not in result
            assert "Line 100" not in result
        finally:
            os.unlink(path)

    def test_read_binary_file_error(self):
        """测试读取二进制文件时的处理"""
        with tempfile.NamedTemporaryFile(
            mode="wb", suffix=".bin", delete=False
        ) as f:
            f.write(bytes(range(256)))
            path = f.name

        try:
            result = read_file(path)
            # 应该有合理的错误处理，不是直接崩溃
            assert isinstance(result, str)
        finally:
            os.unlink(path)

### 19.5 调试日志系统

```python
"""
调试日志系统。
在开发和调试阶段记录 Agent 的每一步操作。
"""
import logging
import json
import time
from pathlib import Path


class AgentLogger:
    """Agent 专用日志器"""

    def __init__(self, log_file: str = "agent_debug.log", level=logging.DEBUG):
        self.logger = logging.getLogger("agent")
        self.logger.setLevel(level)

        # 文件输出
        handler = logging.FileHandler(log_file, encoding="utf-8")
        handler.setFormatter(
            logging.Formatter(
                "%(asctime)s [%(levelname)s] %(message)s",
                datefmt="%H:%M:%S",
            )
        )
        self.logger.addHandler(handler)

        self._call_log: list[dict] = []

    def log_request(self, messages_count: int, tools_count: int):
        """记录 API 请求"""
        self.logger.debug(
            f"API 请求: {messages_count} 条消息, {tools_count} 个工具"
        )

    def log_response(self, finish_reason: str, tokens_used: int = 0):
        """记录 API 响应"""
        self.logger.debug(
            f"API 响应: finish_reason={finish_reason}, tokens={tokens_used}"
        )

    def log_tool_call(self, name: str, args: dict):
        """记录工具调用"""
        entry = {
            "type": "tool_call",
            "name": name,
            "args": args,
            "timestamp": time.time(),
        }
        self._call_log.append(entry)
        self.logger.info(f"工具调用: {name}({json.dumps(args, ensure_ascii=False)})")

    def log_tool_result(self, name: str, result_length: int, duration_ms: float):
        """记录工具执行结果"""
        self.logger.info(
            f"工具结果: {name}, {result_length} 字符, 耗时 {duration_ms:.0f}ms"
        )

    def log_compression(self, strategy: str, before_tokens: int, after_tokens: int):
        """记录上下文压缩"""
        self.logger.info(
            f"上下文压缩: strategy={strategy}, "
            f"{before_tokens} -> {after_tokens} tokens"
        )

    def get_call_summary(self) -> str:
        """获取本次会话的工具调用摘要"""
        if not self._call_log:
            return "本次会话无工具调用"

        tool_counts = {}
        for entry in self._call_log:
            name = entry["name"]
            tool_counts[name] = tool_counts.get(name, 0) + 1

        lines = ["工具调用统计:"]
        for name, count in sorted(tool_counts.items()):
            lines.append(f"  {name}: {count} 次")
        lines.append(f"  总计: {len(self._call_log)} 次")
        return "\n".join(lines)
```

## 常见问题

### Q1: Mock 响应的结构和真实 API 响应不一致怎么办？

定期对比 Mock 响应结构和真实 API 返回结构。一个实用技巧：在开发初期先用真实 API 调试一次，把真实响应打印出来，然后基于真实响应构造 Mock。使用 MagicMock 的灵活属性访问特性可以避免遗漏字段。

### Q2: 异步代码的 Mock 测试怎么写？

使用 `pytest-asyncio` 和 `AsyncMock`。将测试函数声明为 `async def`，加上 `@pytest.mark.asyncio` 装饰器。异步函数的 Mock 用 `AsyncMock(return_value=...)` 代替 `MagicMock(return_value=...)`。

### Q3: 如何测试流式响应的处理逻辑？

使用 `MockFactory.stream_chunks()` 生成 chunk 序列，将 API 的返回值 mock 为一个迭代器（`iter(chunks)`）。在测试中逐个收集 chunk，验证拼接后的完整文本是否正确。

### Q4: 测试覆盖率目标是多少？

工具函数的目标是 90%+，因为逻辑确定性强。Agent Loop 的目标可以低一些（70-80%），因为涉及 Mock 链路较复杂。关键是覆盖所有错误路径（文件不存在、API 超时、工具调用失败等），而不是追求行覆盖率数字。

## 本章小结

本章建立了完整的 Agent 测试体系。`MockFactory` 是核心基础设施，它能快速构造各种类型的 Mock LLM 响应（文本回复、工具调用、流式输出、错误响应），使测试变得确定、快速、零成本。测试分为三层：工具函数的单元测试验证基本逻辑正确性；Provider 的 Mock 测试验证响应解析和错误处理；Agent Loop 的端到端 Mock 测试验证完整的工具调用循环。`AgentLogger` 提供了调试阶段的可观测性。这套测试体系确保 Agent 的每次代码修改都不会引入回归问题。
