```jim
fun fib(n) {
  if (n <= 1) return n;
  return fib(n - 1) + fib(n - 2);
}

print("fib(10) = " + str(fib(10)));
// 输出: fib(10) = 55

// 性能测试

print("耗时: " + str(elapsed) + "s");
```

### 程序二：冒泡排序（验证数组、循环、比较）

```jim
fun bubbleSort(arr) {
  var n = len(arr);
  var i = 0;
  while (i < n - 1) {
    var j = 0;
    while (j < n - i - 1) {
      if (arr[j] > arr[j + 1]) {
        // 交换
        var temp = arr[j];
        arr[j] = arr[j + 1];
        arr[j + 1] = temp;
      }
      j = j + 1;
    }
    i = i + 1;
  }
  return arr;
}

var data = [64, 34, 25, 12, 22, 11, 90];
print("排序前: " + str(data));
bubbleSort(data);
print("排序后: " + str(data));
// 输出: 排序前: [64, 34, 25, 12, 22, 11, 90]
//       排序后: [11, 12, 22, 25, 34, 64, 90]
```

### 程序三：链表 OOP 实现（验证类、方法、对象引用）

```jim
class Node {
  init(value) {
    this.value = value;
    this.next = nil;
  }
}

class LinkedList {
  init() {
    this.head = nil;
    this.size = 0;
  }

  push(value) {
    var node = Node(value);
    node.next = this.head;
    this.head = node;
    this.size = this.size + 1;
  }

  pop() {
    if (this.head == nil) {
      print("链表为空");
      return nil;
    }
    var value = this.head.value;
    this.head = this.head.next;
    this.size = this.size - 1;
    return value;
  }

  contains(value) {
    var cur = this.head;
    while (cur != nil) {
      if (cur.value == value) return true;
      cur = cur.next;
    }
    return false;
  }

  print() {
    var cur = this.head;
    var result = "[";
    while (cur != nil) {
      result = result + str(cur.value);
      if (cur.next != nil) result = result + ", ";
      cur = cur.next;
    }
    result = result + "]";
    print(result);
  }
}

var list = LinkedList();
list.push(3);
list.push(2);
list.push(1);
list.print();        // 输出: [1, 2, 3]
print("size: " + str(list.size));  // 输出: size: 3
print("pop: " + str(list.pop()));  // 输出: pop: 1
list.print();        // 输出: [2, 3]
print("contains 2: " + str(list.contains(2)));  // true
print("contains 5: " + str(list.contains(5)));  // false
```

### 解释器 vs VM 性能对比

```
对比方法：
  1. 用同一份 JimLang 程序
  2. 切换 useVM 标志
  3. 用 clock() 测量执行时间
  4. 重复 10 次取平均值

| 程序类型     | 解释器 | VM   | 加速比 | 原因 |
|------------|--------|------|--------|------|
| fib(30)    | ~800ms | ~200ms | 4x   | 栈帧开销减少 |
| 循环求和 1M | ~500ms | ~80ms | 6x   | 指令循环紧凑 |
| 对象密集   | ~600ms | ~400ms | 1.5x | 对象分配开销相同 |
| 字符串处理  | ~300ms | ~100ms | 3x   | 常量池缓存 |
```

### 性能差异根因分析

```
解释器慢在哪里？

1. 虚方法分发：
   AST 节点通过 accept(visitor) 分发
   → 每个节点一次虚方法调用（~10ns）
   → 100 万次循环 = 10ms 额外开销

2. 内存访问模式：
   AST 节点分散在堆上
   → 缓存命中率低
   → 频繁 cache miss

3. 类型检查：
   运行时频繁 instanceof 检查
   → 每次执行都需要判断节点类型

VM 快在哪里？

1. 紧凑指令：
   1 字节 OpCode + 1 字节操作数
   → 连续内存，缓存友好

2. switch 分发：
   JVM 对 switch 有优化（tableswitch）
   → 比虚方法调用更快

3. 栈式计算：
   操作数栈替代 AST 遍历
   → 减少方法调用层级
```

## 关键代码

```java
public class ProgramRunner {

    public static void runWithInterpreter(String source) {
        Lexer lexer = new Lexer(source);
        Parser parser = new Parser(lexer.tokenize());
        List<Stmt> ast = parser.parse();
        Interpreter interpreter = new Interpreter();
        long start = System.nanoTime();
        interpreter.execute(ast);
        long elapsed = System.nanoTime() - start;
        System.out.printf("解释器模式耗时: %.2f ms%n", elapsed / 1e6);
    }

    public static void runWithVM(String source) {
        Lexer lexer = new Lexer(source);
        Parser parser = new Parser(lexer.tokenize());
        List<Stmt> ast = parser.parse();
        Compiler compiler = new Compiler();
        Chunk chunk = compiler.compile(ast);
        VM vm = new VM();
        long start = System.nanoTime();
        vm.execute(chunk);
        long elapsed = System.nanoTime() - start;
        System.out.printf("VM 模式耗时: %.2f ms%n", elapsed / 1e6);
    }
}
```

## 常见问题 Q&A

**Q1: VM 为什么比解释器快 3-6 倍？**

A: 解释器每次执行需要递归遍历 AST 节点（大量对象分配和方法调用）。VM 执行的是紧凑的字节码数组，缓存友好且无对象分配开销。

**Q2: 为什么不是 10 倍以上？**

A: 我们的 VM 仍是纯 Java 解释执行字节码。真正的高性能需要 JIT（将热点字节码编译为机器码），那才能达到 10-100 倍加速。

**Q3: 为什么测试这三类程序？**

A: 递归测试函数调用机制，循环测试控制流效率，OOP 测试对象系统。三类覆盖了语言的核心运行时。
